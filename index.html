ตั้งค่าลิงก์ระบบหลักและข้อมูลติดต่อ

วิธีสร้าง Google Sheet หลังบ้านใน 2 นาที (ทำครั้งเดียว)
1. สร้าง Google Sheet ใหม่ 1 ไฟล์ → เมนู ส่วนขยาย (Extensions) > Apps Script
2. กดปุ่ม "คัดลอกโค้ด Apps Script" ด้านขวานี้ ไปวางทับใน Apps Script แล้วกด Deploy > New deployment > Web app (Execute as: Me, Who has access: Anyone)
3. นำลิงก์ /exec ที่ได้มาวางในช่องด้านบน แล้วกด "ส่งข้อมูลปัจจุบันทั้งหมดขึ้นสร้างตารางใน Sheet" ระบบจะสร้างแท็บและกรอกข้อมูลทั้ง 15 เอกสาร + 182 รายชื่อลงใน Sheet ให้ทันที! 
/**
 * หลังบ้านเว็บงานแผนงาน โรงเรียนพิษณุโลกพิทยาคม (Google Sheet + Apps Script API)
 * --------------------------------------------------------------------------
 * วิธีติดตั้ง (ใช้เวลาประมาณ 2 นาที):
 * 1. สร้าง Google Sheet ใหม่ 1 ไฟล์ ตั้งชื่อเช่น "หลังบ้านเว็บแผนงาน พ.พ."
 * 2. ไปที่เมนู ส่วนขยาย (Extensions) > Apps Script
 * 3. ลบโค้ดเดิมออกทั้งหมด แล้ววางโค้ดไฟล์นี้ลงไป กดบันทึก (Ctrl+S / Cmd+S)
 * 4. เปลี่ยนรหัสผ่านแอดมินที่บรรทัด ADMIN_PIN ด้านล่าง (ค่าเริ่มต้นคือ "pp2570")
 * 5. กดปุ่ม "ทำให้ใช้งานได้" (Deploy) > "การทำให้ใช้งานได้รายการใหม่" (New deployment)
 *    - ประเภท: เว็บแอป (Web app)
 *    - ดำเนินการในฐานะ: ฉัน (Me)
 *    - ผู้ที่มีสิทธิ์เข้าถึง: ทุกคน (Anyone)  <-- เพื่อให้หน้าเว็บโหลดข้อมูลได้โดยคนเปิดไม่ต้องล็อกอิน
 * 6. คัดลอก URL ของเว็บแอป (ลงท้ายด้วย /exec) ไปวางในหน้าแอดมินของเว็บ (#admin)
 *    แล้วกดปุ่ม "ส่งข้อมูลเริ่มต้นขึ้น Google Sheet" บนหน้าเว็บได้เลย!
 */

const ADMIN_PIN = 'pp2570'; // <-- เปลี่ยนรหัสผ่านทีมแผนงานตรงนี้ได้ตามต้องการ

const SHEET_SETTINGS = 'ตั้งค่าเว็บ';
const SHEET_DOCS     = 'เอกสารและลิงก์';
const SHEET_STAFF    = 'ผู้รับผิดชอบ';

/**
 * เปิดอ่านข้อมูลทั้งหมด (GET) — หน้าเว็บจะเรียกอัตโนมัติเมื่อมีคนเปิดเว็บ
 */
function doGet(e) {
  try {
    const ss = SpreadsheetApp.getActiveSpreadsheet();
    ensureSheetsExist_(ss);
    const data = readAllFromSheets_(ss);
    return jsonResponse_({ ok: true, data: data });
  } catch (err) {
    return jsonResponse_({ ok: false, error: String(err) });
  }
}

/**
 * รับคำสั่งแก้ไขจากหน้าแอดมินบนเว็บ (POST)
 * ใช้ Content-Type: text/plain เพื่อไม่ติด CORS preflight จาก GitHub Pages หรือไฟล์ในเครื่อง
 */
function doPost(e) {
  try {
    const raw = (e && e.postData && e.postData.contents) ? e.postData.contents : '{}';
    const body = JSON.parse(raw);

    if (String(body.pin || '').trim() !== ADMIN_PIN) {
      return jsonResponse_({ ok: false, error: 'รหัสผ่านแอดมินไม่ถูกต้อง' });
    }

    const ss = SpreadsheetApp.getActiveSpreadsheet();
    ensureSheetsExist_(ss);
    const action = body.action || '';

    if (action === 'ping') {
      return jsonResponse_({ ok: true, message: 'เชื่อมต่อสำเร็จ', data: readAllFromSheets_(ss) });
    }

    if (action === 'init_all' || action === 'save_all') {
      if (body.settings) writeSettingsSheet_(ss, body.settings);
      if (Array.isArray(body.documents)) writeDocsSheet_(ss, body.documents);
      if (Array.isArray(body.staff)) writeStaffSheet_(ss, body.staff);
      touchUpdatedDate_(ss);
      return jsonResponse_({ ok: true, data: readAllFromSheets_(ss) });
    }

    if (action === 'save_settings') {
      writeSettingsSheet_(ss, body.settings || {});
      touchUpdatedDate_(ss);
      return jsonResponse_({ ok: true, data: readAllFromSheets_(ss) });
    }

    if (action === 'save_documents') {
      writeDocsSheet_(ss, body.documents || []);
      touchUpdatedDate_(ss);
      return jsonResponse_({ ok: true, data: readAllFromSheets_(ss) });
    }

    if (action === 'save_staff') {
      writeStaffSheet_(ss, body.staff || []);
      touchUpdatedDate_(ss);
      return jsonResponse_({ ok: true, data: readAllFromSheets_(ss) });
    }

    return jsonResponse_({ ok: false, error: 'ไม่รู้จักคำสั่ง: ' + action });
  } catch (err) {
    return jsonResponse_({ ok: false, error: String(err) });
  }
}

/* ==================== HELPER FUNCTIONS ==================== */

function jsonResponse_(obj) {
  return ContentService
    .createTextOutput(JSON.stringify(obj))
    .setMimeType(ContentService.MimeType.JSON);
}

function ensureSheetsExist_(ss) {
  if (!ss.getSheetByName(SHEET_DOCS)) {
    const s = ss.insertSheet(SHEET_DOCS);
    s.appendRow(['id', 'category', 'fy', 'title', 'subtitle', 'url', 'badge', 'featured_home']);
    s.getRange(1, 1, 1, 8).setFontWeight('bold').setBackground('#f3edf7');
    s.setFrozenRows(1);
  }
  if (!ss.getSheetByName(SHEET_STAFF)) {
    const s = ss.insertSheet(SHEET_STAFF);
    s.appendRow(['no', 'name', 'department', 'workgroup']);
    s.getRange(1, 1, 1, 4).setFontWeight('bold').setBackground('#f3edf7');
    s.setFrozenRows(1);
  }
  if (!ss.getSheetByName(SHEET_SETTINGS)) {
    const s = ss.insertSheet(SHEET_SETTINGS);
    s.appendRow(['key', 'value', 'note']);
    s.getRange(1, 1, 1, 3).setFontWeight('bold').setBackground('#f3edf7');
    s.setFrozenRows(1);
  }
}

function readAllFromSheets_(ss) {
  // 1. Settings
  const sSet = ss.getSheetByName(SHEET_SETTINGS);
  const setRows = sSet.getDataRange().getValues();
  const settings = {};
  for (let i = 1; i < setRows.length; i++) {
    const k = String(setRows[i][0] || '').trim();
    const v = String(setRows[i][1] || '').trim();
    if (k) settings[k] = v;
  }

  // 2. Documents
  const sDoc = ss.getSheetByName(SHEET_DOCS);
  const docRows = sDoc.getDataRange().getValues();
  const documents = [];
  for (let i = 1; i < docRows.length; i++) {
    const r = docRows[i];
    const title = String(r[3] || '').trim();
    const url = String(r[5] || '').trim();
    if (!title && !url) continue;
    const featVal = String(r[7] || '').trim().toLowerCase();
    documents.push({
      id: String(r[0] || ('doc-' + i)).trim(),
      category: String(r[1] || 'forms').trim(),
      fy: String(r[2] || '2570').trim(),
      title: title,
      subtitle: String(r[4] || '').trim(),
      url: url,
      badge: String(r[6] || 'เปิดเอกสาร').trim(),
      featured_home: (r[7] === true || featVal === 'true' || featVal === '1' || featVal === 'ใช่' || featVal === 'yes')
    });
  }

  // 3. Staff (Rule L-5: strip any email/@username automatically)
  const sStaff = ss.getSheetByName(SHEET_STAFF);
  const staffRows = sStaff.getDataRange().getValues();
  const staff = [];
  for (let i = 1; i < staffRows.length; i++) {
    const r = staffRows[i];
    const name = sanitizeNoEmail_(String(r[1] || '').trim());
    if (!name) continue;
    const deptVal = sanitizeNoEmail_(String(r[2] || '').trim());
    const grpVal = sanitizeNoEmail_(String(r[3] || '').trim());
    staff.push({
      no: String(r[0] || staff.length + 1).trim(),
      name: name,
      dept: deptVal,
      group: grpVal,
      department: deptVal,
      workgroup: grpVal
    });
  }

  return {
    settings: settings,
    documents: documents,
    staff: staff
  };
}

function sanitizeNoEmail_(text) {
  // ป้องกันการเผลอวางอีเมลในช่องชื่อหรือกลุ่มงาน (กติกา L-5)
  return String(text || '').replace(/[A-Za-z0-9._%+-]+@[A-Za-z0-9.-]+\.[A-Za-z]{2,}/g, '').trim();
}

function writeSettingsSheet_(ss, settings) {
  const s = ss.getSheetByName(SHEET_SETTINGS);
  s.clearContents();
  const notes = {
    school_name: 'ชื่อโรงเรียน',
    office_name: 'ชื่อฝ่ายงาน',
    affiliation: 'สังกัด',
    contact_email: 'อีเมลติดต่อฝ่ายแผนงาน',
    school_site_url: 'ลิงก์เว็บไซต์หลักโรงเรียน',
    ppksar_url: 'ลิงก์ปุ่มหลัก 1: ระบบแผนงาน (ppksar)',
    status_2570_url: 'ลิงก์ปุ่มหลัก 2: เว็บตรวจสถานะกิจกรรม 2570',
    fy_override: 'ปีงบประมาณปัจจุบัน (เช่น 2570 หรือเว้นว่างให้คำนวณอัตโนมัติ)',
    updated_date: 'วันที่อัปเดตข้อมูลล่าสุด'
  };
  const rows = [['key', 'value', 'note']];
  Object.keys(notes).forEach(function(k) {
    rows.push([k, settings[k] !== undefined ? String(settings[k]) : '', notes[k]]);
  });
  Object.keys(settings).forEach(function(k) {
    if (!notes.hasOwnProperty(k)) {
      rows.push([k, String(settings[k]), '']);
    }
  });
  s.getRange(1, 1, rows.length, 3).setValues(rows);
  s.getRange(1, 1, 1, 3).setFontWeight('bold').setBackground('#f3edf7');
}

function writeDocsSheet_(ss, docs) {
  const s = ss.getSheetByName(SHEET_DOCS);
  s.clearContents();
  const rows = [['id', 'category', 'fy', 'title', 'subtitle', 'url', 'badge', 'featured_home']];
  docs.forEach(function(d, idx) {
    if (!d.title && !d.url) return;
    rows.push([
      String(d.id || ('doc-' + (idx + 1))),
      String(d.category || 'forms'),
      String(d.fy || '2570'),
      String(d.title || ''),
      String(d.subtitle || ''),
      String(d.url || ''),
      String(d.badge || 'เปิดเอกสาร'),
      d.featured_home ? 'TRUE' : 'FALSE'
    ]);
  });
  s.getRange(1, 1, rows.length, 8).setValues(rows);
  s.getRange(1, 1, 1, 8).setFontWeight('bold').setBackground('#f3edf7');
}

function writeStaffSheet_(ss, staffList) {
  const s = ss.getSheetByName(SHEET_STAFF);
  s.clearContents();
  const rows = [['no', 'name', 'department', 'workgroup']];
  staffList.forEach(function(p, idx) {
    const name = sanitizeNoEmail_(p.name);
    if (!name) return;
    rows.push([
      String(p.no || (idx + 1)),
      name,
      sanitizeNoEmail_(p.dept || p.department || ''),
      sanitizeNoEmail_(p.group || p.workgroup || '')
    ]);
  });
  s.getRange(1, 1, rows.length, 4).setValues(rows);
  s.getRange(1, 1, 1, 4).setFontWeight('bold').setBackground('#f3edf7');
}

function touchUpdatedDate_(ss) {
  const months = ['มกราคม','กุมภาพันธ์','มีนาคม','เมษายน','พฤษภาคม','มิถุนายน','กรกฎาคม','สิงหาคม','กันยายน','ตุลาคม','พฤศจิกายน','ธันวาคม'];
  const now = new Date();
  const thaiDate = now.getDate() + ' ' + months[now.getMonth()] + ' ' + (now.getFullYear() + 543);
  const s = ss.getSheetByName(SHEET_SETTINGS);
  const rows = s.getDataRange().getValues();
  for (let i = 1; i < rows.length; i++) {
    if (String(rows[i][0]) === 'updated_date') {
      s.getRange(i + 1, 2).setValue(thaiDate);
      return;
    }
  }
  s.appendRow(['updated_date', thaiDate, 'วันที่อัปเดตข้อมูลล่าสุด']);
}
